-- // Safe clipboard
local function copyLink(text)
    local success, err = pcall(function()
        if setclipboard then
            setclipboard(text)
        elseif toclipboard then
            toclipboard(text)
        else
            print("Link: " .. text)
        end
    end)
    if not success then
        print("Clipboard failed:", err)
    end
end
